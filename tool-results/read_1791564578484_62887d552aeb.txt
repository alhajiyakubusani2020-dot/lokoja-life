    55→/* ------------------------------- store -------------------------------- */
    56→
    57→interface GameStore {
    58→  /* UI-level state */
    59→  screen: Screen
    60→  booted: boolean
    61→  hasSave: boolean
    62→  notifications: GameNotification[]
    63→
    64→  /* live game state (null until a game starts) */
    65→  player: PlayerState | null
    66→  world: WorldState | null
    67→  settings: GameSettings
    68→
    69→  /* lifecycle */
    70→  boot: () => void
    71→  setScreen: (screen: Screen) => void
    72→
    73→  /* core actions */
    74→  startNewGame: (name: string, gender: Gender, avatar: AvatarConfig) => void
    75→  continueGame: () => boolean
    76→  returnToMenu: () => void
    77→
    78→  /* needs / time actions (Stage 1: rest; Stage 2: travel — both use the same clock) */
    79→  rest: (kind: 'nap' | 'sleep') => void
    80→  advanceTime: (slots?: number) => void
    81→
    82→  /* travel (Stage 2) */
    83→  isTravelling: boolean
    84→  travel: (destinationId: string, transport: TransportId) => void
    85→
    86→  /* work & opportunities (Stage 3) */
    87→  /** true while a job-shift/hustle resolution transition plays */
    88→  isWorking: boolean
    89→  /** label shown on the working overlay */
    90→  workingLabel: string | null
    91→  /** transient hustle offers found by searching — NOT persisted (gist goes cold) */
    92→  offers: ActiveOpportunity[]
    93→
    94→  /** economy services — every naira flows through these */
    95→  addMoney: (amount: number, meta: { label: string; category: TransactionCategory }) => void
    96→  removeMoney: (amount: number, meta: { label: string; category: TransactionCategory }) => boolean
    97→  giveReward: (reward: Reward, meta: { label: string; category: TransactionCategory }) => void
    98→  addSkillXp: (skill: SkillId, amount: number) => void
    99→
   100→  /** formal jobs (source: board) */
   101→  workShift: (jobId: string) => void
   102→  /** hustle lifecycle: find → accept → do (or drop) — same shape the phone will use */
   103→  searchHustles: () => void
   104→  acceptOffer: (instanceId: string) => void
   105→  declineOffer: (instanceId: string) => void
   106→  completeOpportunity: (instanceId: string) => void
   107→  dropOpportunity: (instanceId: string) => void
   108→  /** remove offers whose time is up (called lazily by the work screen) */
   109→  pruneExpired: () => void
   110→
   111→  /* engine actions used by all future systems */
   112→  addCash: (amount: number) => void
   113→  addXp: (amount: number) => void
   114→  addReputation: (amount: number) => void
   115→
   116→  /* settings & save */
   117→  toggleSound: () => void
   118→  saveNow: () => void
   119→  resetGame: () => void
   120→
   121→  /* feedback */
   122→  notify: (n: Omit<GameNotification, 'id'>) => void
   123→  dismissNotification: (id: number) => void
   124→}

   126→let saveTimer: ReturnType<typeof setTimeout> | null = null
   127→let notificationId = 0
   128→
   129→/** Ledger cap — newest entries survive, old ones fall off. */
   130→const LEDGER_CAP = 50
   131→
   132→/** Debounced autosave — called after every state-mutating action. */
   133→function scheduleAutosave(get: () => GameStore) {
   134→  if (saveTimer) clearTimeout(saveTimer)
   135→  saveTimer = setTimeout(() => {
   136→    const { player, world, settings } = get()
   137→    if (!player || !world) return
   138→    const existing = SaveSystem.load()
   139→    SaveSystem.write({
   140→      version: SaveSystem.SAVE_VERSION,
   141→      createdAt: existing?.createdAt ?? Date.now(),
   142→      updatedAt: Date.now(),
   143→      player,
   144→      world,
   145→      settings,
   146→    })
   147→  }, AUTOSAVE.debounceMs)
   148→}

   150→export const useGameStore = create<GameStore>((set, get) => ({
   151→  screen: 'menu',
   152→  booted: false,
   153→  hasSave: false,
   154→  notifications: [],
   155→
   156→  player: null,
   157→  world: null,
   158→  settings: { sound: true },
   159→
   160→  isTravelling: false,
   161→  isWorking: false,
   162→  workingLabel: null,
   163→  offers: [],
   164→
   165→  /* ----------------------------- lifecycle ----------------------------- */
   166→
   167→  boot: () => {
   168→    const save = SaveSystem.load()
   169→    syncAudioWithSettings(save?.settings ?? { sound: true })
   170→    set({ booted: true, hasSave: SaveSystem.hasSave(), settings: save?.settings ?? { sound: true } })
   171→  },
   172→
   173→  setScreen: (screen) => set({ screen }),
   174→
   175→  /* ---------------------------- core actions ---------------------------- */
   176→
   177→  startNewGame: (name, gender, avatar) => {
   178→    const player: PlayerState = {
   179→      name: name.trim() || 'New Hustler',
   180→      gender,
   181→      avatar: { ...DEFAULT_AVATAR, ...avatar },
   182→      cash: STARTING.cash,
   183→      needs: { ...STARTING.needs },
   184→      skills: { ...STARTING.skills },
   185→      reputation: STARTING.reputation,
   186→      level: STARTING.level,
   187→      xp: STARTING.xp,
   188→      homeId: STARTING.homeId,
   189→      jobId: STARTING.jobId,
   190→      skillXp: { hustle: 0, smart: 0, social: 0 },
   191→      activeOpportunities: [],
   192→      cooldowns: {},
   193→      transactions: [],
   194→      stats: { jobsCompleted: 0, hustlesCompleted: 0, totalEarned: 0, totalSpent: 0 },
   195→      inventory: [],
   196→      businesses: [],
   197→      missions: {},
   198→      achievements: [],
   199→      discoveries: [],
   200→    }
   201→    const world: WorldState = {
   202→      day: 1,
   203→      timeOfDay: 'morning',
   204→      weather: 'sunny',
   205→      currentLocation: 'home',
   206→    }
   207→    set({ player, world, screen: 'game', offers: [] })
   208→    scheduleAutosave(get)
   209→  },
   210→
   211→  continueGame: () => {
   212→    const save = SaveSystem.load()
   213→    if (!save) return false
   214→    syncAudioWithSettings(save.settings)
   215→    set({ player: save.player, world: save.world, settings: save.settings, screen: 'game' })
   216→    return true
   217→  },
   218→
   219→  returnToMenu: () => {
   220→    get().saveNow()
   221→    set({ screen: 'menu', hasSave: SaveSystem.hasSave() })
   222→  },
   223→
   224→  /* --------------------------- needs / time ----------------------------- */
   225→
   226→  rest: (kind) => {
   227→    const { player, world } = get()
   228→    if (!player || !world) return
   229→
   230→    // rest values come from the central config — never hard-coded
   231→    const { energy, hunger, mood, health } = NEEDS.rest[kind]
   232→    const delta = { energy, hunger, mood, health }
   233→
   234→    const needs = applyNeedsDelta(player.needs, delta)
   235→
   236→    let newWorld: WorldState
   237→    if (kind === 'nap') {
   238→      const { timeOfDay, dayDelta } = nextSlot(world.timeOfDay)
   239→      newWorld = {
   240→        ...world,
   241→        timeOfDay,
   242→        day: world.day + dayDelta,
   243→        weather: weatherForSlot(world.weather, timeOfDay),
   244→      }
   245→    } else {
   246→      // sleep till next morning — fresh day, fresh weather
   247→      newWorld = {
   248→        ...world,
   249→        day: world.day + 1,
   250→        timeOfDay: 'morning',
   251→        weather: rollNewDayWeather(),
   252→      }
   253→    }
   254→
   255→    set({ player: { ...player, needs }, world: newWorld })
   256→    get().notify({
   257→      icon: kind === 'nap' ? '😴' : '🌙',
   258→      text: kind === 'nap' ? 'You took a short nap.' : `Day ${newWorld.day} begins!`,
   259→      tone: 'good',
   260→    })
   261→    audio.play('notify')
   262→    scheduleAutosave(get)
   263→  },
   264→
   265→  advanceTime: (slots = 1) => {
   266→    const { world } = get()
   267→    if (!world) return
   268→    const clock = advanceWorldClock(world.timeOfDay, world.day, world.weather, slots)
   269→    set({ world: { ...world, ...clock } })
   270→    scheduleAutosave(get)
   271→  },
   272→
   273→  /* ------------------------------- travel ------------------------------- */
   274→
   275→  travel: (destinationId, transportId) => {
   276→    const { player, world, isTravelling } = get()
   277→    if (!player || !world || isTravelling) return
   278→
   279→    const origin = LOCATION_BY_ID[world.currentLocation]
   280→    const destination = LOCATION_BY_ID[destinationId]
   281→    if (!origin || !destination || origin.id === destination.id) return
   282→
   283→    // --- guards (all feedback through notifications, never silent) ---
   284→    if (player.level < destination.unlockLevel) {
   285→      audio.play('notify')
   286→      get().notify({
   287→        icon: '🔒',
   288→        text: `${destination.name} unlocks at Level ${destination.unlockLevel}.`,
   289→        tone: 'bad',
   290→      })
   291→      return
   292→    }
   293→    if (!destination.transport.includes(transportId)) return
   294→
   295→    // pure calculation — same inputs always give the same plan
   296→    const plan = calculateTravel(origin, destination, transportId, world.weather, world.timeOfDay)
   297→    const event = rollTravelEvent(transportId, world.weather)
   298→
   299→    if (player.cash < plan.cost) {
   300→      audio.play('notify')
   301→      get().notify({
   302→        icon: '💸',
   303→        text: `The fare is ${formatNaira(plan.cost)} — you no get am.`,
   304→        tone: 'bad',
   305→      })
   306→      return
   307→    }
   308→    if (player.needs.energy < plan.energyCost) {
   309→      audio.play('notify')
   310→      get().notify({
   311→        icon: '🥵',
   312→        text: 'You are too tired for that journey. Rest first.',
   313→        tone: 'bad',
   314→      })
   315→      return
   316→    }
   317→
   318→    // travelling state drives the 📍 HUD + map transition
   319→    set({ isTravelling: true })
   320→
   321→    // apply the completed trip on the EXISTING clock through proper systems
   322→    const applyArrival = () => {
   323→      const { player: p, world: w } = get()
   324→      if (!p || !w) return
   325→
   326→      const cash = Math.max(0, p.cash - plan.cost + (event?.cashDelta ?? 0))
   327→      const needs = applyNeedsDelta(p.needs, {
   328→        energy: -(plan.energyCost + (event?.energyDelta ?? 0)),
   329→        mood: event?.moodDelta ?? 0,
   330→      })
   331→      const clock = advanceWorldClock(w.timeOfDay, w.day, w.weather, plan.slots)
   332→
   333→      // Stage 3: record the fare (and any event tip) in the player's ledger
   334→      const arrivalSlot = absoluteSlot(clock.day, clock.timeOfDay)
   335→      const txs: Transaction[] = []
   336→      if (plan.cost > 0) {
   337→        txs.push(
   338→          recordTransaction(`Fare to ${destination.name}`, 'transport', -plan.cost, clock.day, arrivalSlot),
   339→        )
   340→      }
   341→      if (event?.cashDelta && event.cashDelta > 0) {
   342→        txs.push(recordTransaction(event.title, 'tip', event.cashDelta, clock.day, arrivalSlot))
   343→      }
   344→      const stats = {
   345→        ...p.stats,
   346→        totalEarned:
   347→          p.stats.totalEarned + (event?.cashDelta && event.cashDelta > 0 ? event.cashDelta : 0),
   348→        totalSpent: p.stats.totalSpent + plan.cost,
   349→      }
   350→
   351→      set({
   352→        player: {
   353→          ...p,
   354→          cash,
   355→          needs,
   356→          transactions: [...txs, ...p.transactions].slice(0, LEDGER_CAP),
   357→          stats,
   358→        },
   359→        world: { ...w, currentLocation: destination.id, ...clock },
   360→        isTravelling: false,
   361→        offers: [], // left the area — unclaimed gist goes cold
   362→      })
   363→
   364→      // arrival feedback: destination + a compact time/money/energy summary
   365→      get().notify({
   366→        icon: destination.emoji,
   367→        text: `You arrived at ${destination.name}.`,
   368→        tone: 'neutral',
   369→      })
   370→      get().notify({
   371→        icon: '🧾',
   372→        text: `${formatNaira(cash)} • ⚡ ${Math.round(needs.energy)}`,
   373→        tone: 'neutral',
   374→      })
   375→      if (event) {
   376→        get().notify({ icon: event.icon, text: event.description, tone: event.tone })
   377→        if (event.xpDelta && event.xpDelta > 0) get().addXp(event.xpDelta)
   378→      }
   379→      audio.play('notify')
   380→      scheduleAutosave(get)
   381→    }
   382→
   383→    setTimeout(applyArrival, TRAVEL.transitionMs)
   384→  },
   385→
   386→  /* ------------------------ work & opportunities ------------------------ */
   387→
   388→  addMoney: (amount, meta) => {
   389→    const { player, world } = get()
   390→    if (!player || !world || amount <= 0) return
   391→    const slot = absoluteSlot(world.day, world.timeOfDay)
   392→    const tx = recordTransaction(meta.label, meta.category, amount, world.day, slot)
   393→    set({
   394→      player: {
   395→        ...player,
   396→        cash: player.cash + amount,
   397→        transactions: pushTransaction(player.transactions, tx, LEDGER_CAP),
   398→        stats: { ...player.stats, totalEarned: player.stats.totalEarned + amount },
   399→      },
   400→    })
   401→    audio.play('chime')
   402→    scheduleAutosave(get)
   403→  },
   404→
   405→  removeMoney: (amount, meta) => {
   406→    const { player, world } = get()
   407→    if (!player || !world || amount <= 0) return false
   408→    if (!canAfford(player.cash, amount)) {
   409→      get().notify({ icon: '💸', text: 'You no get that kind money.', tone: 'bad' })
   410→      return false
   411→    }
   412→    const slot = absoluteSlot(world.day, world.timeOfDay)
   413→    const tx = recordTransaction(meta.label, meta.category, -amount, world.day, slot)
   414→    set({
   415→      player: {
   416→        ...player,
   417→        cash: player.cash - amount,
   418→        transactions: pushTransaction(player.transactions, tx, LEDGER_CAP),
   419→        stats: { ...player.stats, totalSpent: player.stats.totalSpent + amount },
   420→      },
   421→    })
   422→    audio.play('notify')
   423→    scheduleAutosave(get)
   424→    return true
   425→  },
   426→
   427→  /**
   428→   * Apply a full reward bundle (cash + xp + skill xp + reputation) through
   429→   * the proper engine actions, with ONE ledger entry for the cash part.
   430→   * Jobs, hustles, events and future systems all pay through here.
   431→   */
   432→  giveReward: (reward, meta) => {
   433→    const { player, world } = get()
   434→    if (!player || !world) return
   435→
   436→    // 1) cash + ledger entry
   437→    const cash = rewardCash(reward)
   438→    if (cash > 0) {
   439→      const slot = absoluteSlot(world.day, world.timeOfDay)
   440→      const tx = recordTransaction(meta.label, meta.category, cash, world.day, slot)
   441→      set({
   442→        player: {
   443→          ...player,
   444→          cash: player.cash + cash,
   445→          transactions: pushTransaction(player.transactions, tx, LEDGER_CAP),
   446→          stats: { ...player.stats, totalEarned: player.stats.totalEarned + cash },
   447→        },
   448→      })
   449→      audio.play('chime')
   450→    }
   451→
   452→    // 2) player XP (level-ups handled by the existing engine action)
   453→    if (reward.xp && reward.xp > 0) get().addXp(reward.xp)
   454→
   455→    // 3) skill XP
   456→    for (const [skill, amount] of Object.entries(reward.skillXp ?? {})) {
   457→      if (amount && amount > 0) get().addSkillXp(skill as SkillId, amount)
   458→    }
   459→
   460→    // 4) reputation
   461→    if (reward.reputation && reward.reputation !== 0) get().addReputation(reward.reputation)
   462→
   463→    // 5) one compact summary toast
   464→    const summary = describeReward(reward)
   465→    if (summary) get().notify({ icon: '🧾', text: summary, tone: 'good' })
   466→
   467→    scheduleAutosave(get)
   468→  },
   469→
   470→  addSkillXp: (skill, amount) => {
   471→    const { player } = get()
   472→    if (!player || amount <= 0) return
   473→    const result = applySkillXp(player.skillXp, player.skills, player.level, skill, amount)
   474→    set({ player: { ...player, skillXp: result.skillXp, skills: result.skills } })
   475→    if (result.leveledUp) {
   476→      audio.play('chime')
   477→      get().notify({
   478→        icon: SKILL_META[skill].emoji,
   479→        text: `${SKILL_META[skill].label} is now Level ${result.newLevel}!`,
   480→        tone: 'good',
   481→      })
   482→    }
   483→  },
   484→
   485→  /** Work a formal job shift — reliable pay, real time + energy cost. */
   486→  workShift: (jobId) => {
   487→    const { player, world, isWorking, isTravelling } = get()
   488→    if (!player || !world || isWorking || isTravelling) return
   489→
   490→    const job = JOB_BY_ID[jobId]
   491→    if (!job) return
   492→
   493→    // --- guards: location, requirements, cooldown, energy ---
   494→    if (world.currentLocation !== job.locationId) {
   495→      const loc = LOCATION_BY_ID[job.locationId]
   496→      get().notify({
   497→        icon: '📍',
   498→        text: `This shift is at ${loc?.name ?? 'another part of Lokoja'}.`,
   499→        tone: 'bad',
   500→      })
   501→      return
   502→    }
   503→    const req = checkRequirements(player, job.requirements)
   504→    if (!req.ok) {
   505→      audio.play('notify')
   506→      get().notify({ icon: '🔒', text: req.reasons[0], tone: 'bad' })
   507→      return
   508→    }
   509→    const slot = absoluteSlot(world.day, world.timeOfDay)
   510→    const cd = cooldownRemaining(player.cooldowns, job.id, slot)
   511→    if (cd > 0) {
   512→      audio.play('notify')
   513→      get().notify({
   514→        icon: '⏳',
   515→        text: `${job.name} needs a breather — ready in ${cd} time slot${cd > 1 ? 's' : ''}.`,
   516→        tone: 'bad',
   517→      })
   518→      return
   519→    }
   520→    if (player.needs.energy < job.energyCost) {
   521→      audio.play('notify')
   522→      get().notify({ icon: '🥵', text: 'You are too tired for this shift. Rest first.', tone: 'bad' })
   523→      return
   524→    }
   525→
   526→    const result = resolveJob(job, player)
   527→    set({ isWorking: true, workingLabel: job.name })
   528→
   529→    const applyDone = () => {
   530→      const { player: p, world: w } = get()
   531→      if (!p || !w) return
   532→
   533→      const needs = applyNeedsDelta(p.needs, { energy: -job.energyCost })
   534→      const clock = advanceWorldClock(w.timeOfDay, w.day, w.weather, job.timeSlots)
   535→      const afterSlot = absoluteSlot(clock.day, clock.timeOfDay)
   536→      const cooldownSlots = job.cooldownSlots ?? WORK.defaultCooldownSlots
   537→
   538→      set({
   539→        player: {
   540→          ...p,
   541→          needs,
   542→          cooldowns: { ...p.cooldowns, [job.id]: afterSlot + cooldownSlots },
   543→          stats: { ...p.stats, jobsCompleted: p.stats.jobsCompleted + 1 },
   544→        },
   545→        world: { ...w, ...clock },
   546→        isWorking: false,
   547→        workingLabel: null,
   548→      })
   549→
   550→      get().notify({
   551→        icon: result.outcome === 'great' ? '🌟' : '💼',
   552→        text: result.headline,
   553→        tone: 'good',
   554→      })
   555→      get().giveReward(result.reward, { label: job.name, category: 'job' })
   556→      if (result.outcome === 'great') audio.play('levelup')
   557→      scheduleAutosave(get)
   558→    }
   559→
   560→    setTimeout(applyDone, WORK.transitionMs)
   561→  },
   562→
   563→  /** Look around the current area for personal hustles (gist). */
   564→  searchHustles: () => {
   565→    const { player, world } = get()
   566→    if (!player || !world) return
   567→
   568→    const slot = absoluteSlot(world.day, world.timeOfDay)
   569→    if (cooldownRemaining(player.cooldowns, 'hustle-search', slot) > 0) {
   570→      audio.play('notify')
   571→      get().notify({ icon: '⏳', text: 'You just checked around. Give the streets a moment.', tone: 'bad' })
   572→      return
   573→    }
   574→
   575→    const generation = generateHustleOffers(world.currentLocation, player, slot)
   576→    set({
   577→      player: {
   578→        ...player,
   579→        cooldowns: { ...player.cooldowns, 'hustle-search': slot + HUSTLE.searchCooldownSlots },
   580→      },
   581→      offers: generation.offers,
   582→    })
   583→
   584→    if (generation.dry) {
   585→      audio.play('notify')
   586→      get().notify({ icon: '👀', text: 'You looked around but nothing came up this time.', tone: 'neutral' })
   587→    } else {
   588→      audio.play('notify')
   589→      get().notify({
   590→        icon: '🗣️',
   591→        text: `Fresh gist! ${generation.offers.length} offer${generation.offers.length > 1 ? 's' : ''} came up.`,
   592→        tone: 'good',
   593→      })
   594→    }
   595→    scheduleAutosave(get)
   596→  },

   598→  /** Accept a hustle offer — it joins the player's active opportunities. */
   599→  acceptOffer: (instanceId) => {
   600→    const { player, offers } = get()
   601→    if (!player) return
   602→    const offer = offers.find((o) => o.instanceId === instanceId)
   603→    if (!offer) return
   604→
   605→    if (player.activeOpportunities.length >= HUSTLE.maxActive) {
   606→      audio.play('notify')
   607→      get().notify({
   608→        icon: '🎒',
   609→        text: `Your hands are full — finish or drop something first (max ${HUSTLE.maxActive}).`,
   610→        tone: 'bad',
   611→      })
   612→      return
   613→    }
   614→    if (player.activeOpportunities.some((o) => o.defId === offer.defId)) {
   615→      audio.play('notify')
   616→      get().notify({ icon: '🤔', text: 'You already took this one.', tone: 'bad' })
   617→      return
   618→    }
   619→
   620→    set({
   621→      player: { ...player, activeOpportunities: [...player.activeOpportunities, offer] },
   622→      offers: offers.filter((o) => o.instanceId !== instanceId),
   623→    })
   624→    audio.play('click')
   625→    get().notify({ icon: '🤝', text: 'Offer accepted — it goes cold if you wait too long!', tone: 'good' })
   626→    scheduleAutosave(get)
   627→  },
   628→
   629→  declineOffer: (instanceId) => {
   630→    const { offers } = get()
   631→    const offer = offers.find((o) => o.instanceId === instanceId)
   632→    if (!offer) return
   633→    set({ offers: offers.filter((o) => o.instanceId !== instanceId) })
   634→    audio.play('click')
   635→    get().notify({ icon: '🚶', text: 'You declined. No wahala.', tone: 'neutral' })
   636→  },

   638→  /** Do an accepted hustle at its real location — outcome rolls in work.ts. */
   639→  completeOpportunity: (instanceId) => {
   640→    const { player, world, isWorking, isTravelling } = get()
   641→    if (!player || !world || isWorking || isTravelling) return
   642→
   643→    const instance = player.activeOpportunities.find((o) => o.instanceId === instanceId)
   644→    if (!instance) return
   645→    const def = hustleDefFor(instance)
   646→    if (!def) {
   647→      // data changed between versions — drop the orphan quietly but honestly
   648→      set({
   649→        player: {
   650→          ...player,
   651→          activeOpportunities: player.activeOpportunities.filter((o) => o.instanceId !== instanceId),
   652→        },
   653→      })
   654→      return
   655→    }
   656→
   657→    if (world.currentLocation !== instance.locationId) {
   658→      const loc = LOCATION_BY_ID[instance.locationId]
   659→      audio.play('notify')
   660→      get().notify({
   661→        icon: '📍',
   662→        text: `This hustle is at ${loc?.name ?? 'another part of Lokoja'}.`,
   663→        tone: 'bad',
   664→      })
   665→      return
   666→    }
   667→    if (player.needs.energy < def.energyCost) {
   668→      audio.play('notify')
   669→      get().notify({ icon: '🥵', text: 'Too tired for this hustle. Rest small first.', tone: 'bad' })
   670→      return
   671→    }
   672→
   673→    const result = resolveOpportunity(instance, player)
   674→    if (!result) return
   675→    set({ isWorking: true, workingLabel: def.title })
   676→
   677→    const applyDone = () => {
   678→      const { player: p, world: w } = get()
   679→      if (!p || !w) return
   680→
   681→      const needs = applyNeedsDelta(p.needs, {
   682→        energy: -def.energyCost,
   683→        mood: result.outcome === 'fail' ? WORK.fail.moodDelta : 0,
   684→      })
   685→      const clock = advanceWorldClock(w.timeOfDay, w.day, w.weather, def.timeSlots)
   686→      const afterSlot = absoluteSlot(clock.day, clock.timeOfDay)
   687→
   688→      set({
   689→        player: {
   690→          ...p,
   691→          needs,
   692→          cooldowns: { ...p.cooldowns, [def.id]: afterSlot + def.cooldownSlots },
   693→          activeOpportunities: p.activeOpportunities.filter((o) => o.instanceId !== instanceId),
   694→          stats: { ...p.stats, hustlesCompleted: p.stats.hustlesCompleted + 1 },
   695→        },
   696→        world: { ...w, ...clock },
   697→        isWorking: false,
   698→        workingLabel: null,
   699→      })
   700→
   701→      get().notify({
   702→        icon: result.outcome === 'great' ? '🌟' : result.outcome === 'fail' ? '😐' : '🤝',
   703→        text: result.headline,
   704→        tone: result.outcome === 'fail' ? 'bad' : 'good',
   705→      })
   706→      get().giveReward(result.reward, { label: def.title, category: 'hustle' })
   707→      audio.play(result.outcome === 'fail' ? 'notify' : result.outcome === 'great' ? 'levelup' : 'chime')
   708→      scheduleAutosave(get)
   709→    }
   710→
   711→    setTimeout(applyDone, WORK.transitionMs)
   712→  },
   713→
   714→  dropOpportunity: (instanceId) => {
   715→    const { player } = get()
   716→    if (!player) return
   717→    const dropped = player.activeOpportunities.find((o) => o.instanceId === instanceId)
   718→    if (!dropped) return
   719→    const def = hustleDefFor(dropped)
   720→    set({
   721→      player: {
   722→        ...player,
   723→        activeOpportunities: player.activeOpportunities.filter((o) => o.instanceId !== instanceId),
   724→      },
   725→    })
   726→    audio.play('click')
   727→    get().notify({ icon: '🚶', text: `You dropped “${def?.title ?? 'the offer'}”.`, tone: 'neutral' })
   728→    scheduleAutosave(get)
   729→  },

   731→  /** Lazily remove expired offers — the Work screen calls this on mount. */
   732→  pruneExpired: () => {
   733→    const { player, world, offers } = get()
   734→    if (!player || !world) return
   735→    const slot = absoluteSlot(world.day, world.timeOfDay)
   736→
   737→    const { kept, expired } = pruneExpiredOpportunities(player.activeOpportunities, slot)
   738→    const keptOffers = offers.filter((o) => o.expiresAtSlot > slot)
   739→
   740→    if (kept.length === player.activeOpportunities.length && keptOffers.length === offers.length) {
   741→      return
   742→    }
   743→
   744→    set({ player: { ...player, activeOpportunities: kept }, offers: keptOffers })
   745→    for (const o of expired) {
   746→      const def = hustleDefFor(o)
   747→      get().notify({ icon: '⌛', text: `“${def?.title ?? 'An offer'}” went cold.`, tone: 'neutral' })
   748→    }
   749→    scheduleAutosave(get)
   750→  },
