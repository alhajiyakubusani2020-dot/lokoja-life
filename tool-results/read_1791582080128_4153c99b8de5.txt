    35→/** One-line reward preview for a hustle def: ₦range • skill • rep. */
    36→function rewardPreview(def: HustleDef): string {
    37→  const parts = [
    38→    `${formatNaira(def.reward.cashMin)}–${formatNaira(def.reward.cashMax)}`,
    39→    `${SKILL_META[def.skill].emoji}+${def.reward.skillXp}`,
    40→    `⭐+${def.reward.reputation}`,
    41→  ]
    42→  return parts.join(' • ')
    43→}

    45→function SkillPill({ skillId }: { skillId: string }) {
    46→  const meta = SKILL_META[skillId as keyof typeof SKILL_META]
    47→  if (!meta) return null
    48→  return (
    49→    <span className="rounded-full bg-black/[0.05] px-2 py-0.5 text-[10px] font-bold text-[var(--lko-ink-soft)]">
    50→      {meta.emoji} {meta.label}
    51→    </span>
    52→  )
    53→}

    57→function OfferCard({
    58→  instance,
    59→  isHere,
    60→  onDo,
    61→  onDrop,
    62→  onGo,
    63→}: {
    64→  instance: ActiveOpportunity
    65→  isHere: boolean
    66→  onDo: () => void
    67→  onDrop: () => void
    68→  onGo: () => void
    69→}) {
    70→  const player = useGameStore((s) => s.player)!
    71→  const world = useGameStore((s) => s.world)!
    72→  const def = hustleDefFor(instance)
    73→  if (!def) return null
    74→
    75→  const npc = NPC_BY_ID[instance.npcId ?? '']
    76→  const loc = LOCATION_BY_ID[instance.locationId]
    77→  const abs = absoluteSlot(world.day, world.timeOfDay)
    78→  const slotsLeft = instance.expiresAtSlot - abs
    79→  const urgent = slotsLeft <= 2
    80→  const tooTired = player.needs.energy < def.energyCost
    81→  const odds = Math.round(hustleSuccessChance(def, player) * 100)
    82→
    83→  return (
    84→    <div className={`lko-card p-4 ${urgent ? 'border-[var(--lko-terra)]/40' : ''}`}>
    85→      {/* npc header */}
    86→      <div className="flex items-center gap-2.5">
    87→        <span
    88→          className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-[var(--lko-gold-soft)] text-lg"
    89→          aria-hidden
    90→        >
    91→          {npc?.emoji ?? '🙂'}
    92→        </span>
    93→        <div className="min-w-0 flex-1">
    94→          <p className="text-sm font-extrabold leading-tight">{npc?.name ?? 'Someone'}</p>
    95→          <p className="text-[11px] font-semibold text-[var(--lko-ink-soft)]">
    96→            📍 {loc?.name ?? 'Somewhere'} • ⏛ {urgent ? 'going cold soon!' : `${slotsLeft} slot${slotsLeft === 1 ? '' : 's'} left`}
    97→          </p>
    98→        </div>
    99→        <span className="shrink-0 rounded-full bg-[var(--lko-green-soft)] px-2 py-0.5 text-[10px] font-extrabold text-[var(--lko-green-deep)]">
   100→          🎯 {odds}%
   101→        </span>
   102→      </div>
   103→
   104→      {/* the personal message */}
   105→      <p className="mt-2.5 rounded-xl bg-[var(--lko-cream)] px-3 py-2 text-[13px] italic leading-relaxed text-[var(--lko-ink)]">
   106→        “{def.message}”
   107→      </p>
   108→
   109→      {/* terms */}
   110→      <div className="mt-2 flex flex-wrap items-center gap-1.5 text-[11px] font-bold text-[var(--lko-ink-soft)]">
   111→        <span className="rounded-full bg-[var(--lko-gold-soft)] px-2 py-0.5 text-[#8a5a00]">
   112→          {formatNaira(def.reward.cashMin)}–{formatNaira(def.reward.cashMax)}
   113→        </span>
   114→        <SkillPill skillId={def.skill} />
   115→        <span className="rounded-full bg-black/[0.05] px-2 py-0.5">−{def.energyCost} ⚡</span>
   116→        {def.timeSlots > 0 && (
   117→          <span className="rounded-full bg-black/[0.05] px-2 py-0.5">{def.timeSlots} slot</span>
   118→        )}
   119→      </div>
   120→
   121→      {/* actions */}
   122→      <div className="mt-3 grid grid-cols-2 gap-2">
   123→        {isHere ? (
   124→          <button
   125→            onClick={onDo}
   126→            disabled={tooTired}
   127→            className={`lko-press min-h-[46px] rounded-xl border-b-4 px-2 py-2 text-sm font-extrabold ${
   128→              tooTired
   129→                ? 'cursor-not-allowed border-black/10 bg-black/10 text-white/70'
   130→                : 'border-[#0a5c2e] bg-[var(--lko-green)] text-white'
   131→            }`}
   132→          >
   133→            {tooTired ? '🥵 Too tired' : '💪 Do it'}
   134→          </button>
   135→        ) : (
   136→          <button
   137→            onClick={onGo}
   138→            className="lko-press min-h-[46px] rounded-xl border-b-4 border-[#b57d0a] bg-[var(--lko-gold)] px-2 py-2 text-sm font-extrabold text-[#3d2a00]"
   139→          >
   140→            🗺️ Take me there
   141→          </button>
   142→        )}
   143→        <button
   144→          onClick={onDrop}
   145→          className="lko-press min-h-[46px] rounded-xl border-2 border-black/[0.08] bg-white px-2 py-2 text-sm font-bold text-[var(--lko-ink)]"
   146→        >
   147→          Drop
   148→        </button>
   149→      </div>
   150→    </div>
   151→  )
   152→}

   368→  const acceptOffer = useGameStore((s) => s.acceptOffer)
   369→  const declineOffer = useGameStore((s) => s.declineOffer)
   370→  const completeOpportunity = useGameStore((s) => s.completeOpportunity)
   371→  const dropOpportunity = useGameStore((s) => s.dropOpportunity)
   372→  const pruneExpired = useGameStore((s) => s.pruneExpired)
   373→  const workShift = useGameStore((s) => s.workShift)
   374→
   375→  const [sheetJobId, setSheetJobId] = useState<string | null>(null)
   376→
   377→  const location = LOCATION_BY_ID[world.currentLocation]
   378→  const abs = absoluteSlot(world.day, world.timeOfDay)
   379→  const searchCd = cooldownRemaining(player.cooldowns, 'hustle-search', abs)
   380→
   381→  // lazily expire cold offers whenever the work screen is open
   382→  useEffect(() => {
   383→    pruneExpired()
   384→  }, [world.day, world.timeOfDay, offers.length, pruneExpired])
   385→
   386→  const jobsHere = useMemo(() => JOBS.filter((j) => j.locationId === location.id), [location.id])
   387→  const jobsElsewhere = useMemo(() => JOBS.filter((j) => j.locationId !== location.id), [location.id])
   388→
   389→  const todayTxs = transactionsOnDay(player.transactions, world.day)
   390→  const todayEarned = sumEarned(todayTxs)
   391→  const todaySpent = sumSpent(todayTxs)
   392→
   393→  const sheetJob = sheetJobId ? JOBS.find((j) => j.id === sheetJobId) ?? null : null
   394→
   395→  return (
   396→    <div className="px-4 pt-3">
   397→      {/* working transition overlay */}
   398→      {isWorking && (
   399→        <div className="fixed inset-0 z-40 flex items-center justify-center bg-[var(--lko-cream)]/60 backdrop-blur-[2px]">
   400→          <span className="lko-display animate-bounce rounded-full border-2 border-[var(--lko-gold)]/40 bg-white/95 px-5 py-2.5 text-sm shadow-lg">
   401→            💼 Working{workingLabel ? ` — ${workingLabel}` : ''}…
   402→          </span>
   403→        </div>
   404→      )}
   405→
   406→      {/* location header */}
   407→      <div className="lko-card mb-3 flex items-center justify-between gap-2 px-4 py-2.5">
   408→        <div className="min-w-0">
   409→          <p className="text-[11px] font-extrabold uppercase tracking-wider text-[var(--lko-ink-soft)]">
   410→            💼 Work • you dey here
   411→          </p>
   412→          <p className="lko-display truncate text-base">
   413→            {location.emoji} {location.name}
   414→          </p>
   415→        </div>
   416→        <span className="shrink-0 rounded-full bg-[var(--lko-gold-soft)] px-2.5 py-1 text-[11px] font-bold text-[#8a5a00]">
   417→          ⏛ slot {abs % 4 + 1}/4
   418→        </span>
   419→      </div>
   420→
   421→      {/* ------------------------- active offers ------------------------- */}
   422→      {player.activeOpportunities.length > 0 && (
   423→        <section aria-label="Active offers" className="mb-4">
   424→          <h2 className="mb-2 text-xs font-extrabold uppercase tracking-wider text-[var(--lko-ink-soft)]">
   425→            🤝 Your accepted hustles ({player.activeOpportunities.length}/{HUSTLE.maxActive})
   426→          </h2>
   427→          <div className="flex flex-col gap-2.5">
   428→            {player.activeOpportunities.map((o) => (
   429→              <OfferCard
   430→                key={o.instanceId}
   431→                instance={o}
   432→                isHere={o.locationId === world.currentLocation}
   433→                onDo={() => {
   434→                  audio.play('click')
   435→                  completeOpportunity(o.instanceId)

   437→                onDrop={() => {
   438→                  audio.play('click')
   439→                  dropOpportunity(o.instanceId)
   440→                }}

   442→                  audio.play('click')

   444→                }}
   445→              />
   446→            ))}
   447→          </div>
   448→        </section>
